\section{Roasted Spinach and Pumpkin Risotto}

\index{author!The Australian Woman's Weekly}
\index{source!Best Food Mains}
\index{vegetarian}

Source: The Australian Woman's Weekly Best Food Mains Cookbook

\subsection{Ingredients}

\begin{itemize}
	\item 500g pumpkin, chopped coarsely
	\item 2 tablespoons olive oil
	\item 1.25 liters (5 cups) water
	\item 1\sfrac{1}{2} cups (375ml) vegetable stock
	\item 1 large brown onion (200g), chopped coarsely
	\item 2 cloves garlic, crushed
	\item 2 cups (400g) arborio rice
	\item \sfrac{1}{2} cup (125ml) dry white wine
	\item 250g spinach, chopped coarsely
	\item \sfrac{1}{2} cup (80g) toasted pine nuts
	\item \sfrac{1}{2} cup (40g) finely grated parmesan cheese
	\item \sfrac{1}{2} (125ml) cream
\end{itemize}

\subsection{Method}

\begin{enumerate}
	\item Preheat oven to hot.
	\item Combine pumpkin and half of the oil in medium baking dish; roast, uncovered, in hot oven about 20 minutes or until tender.
	\item Meanwhile, combine the water and stock in large saucepan; bring to a boil. Reduce heat; simmer, covered.
	\item Heat remaining oil in large saucepan; cook onion and garlic, stirring, until onion softens. Add rice; stir to coat rice in oil mixture. Add wine; cook, stirring, until liquid is absorbed. Continue adding stock mixture, in 1-cup batches, stirring until absorbed after each addition. Total cooking time should be about 35 minutes or until rice is just tender.
	\item Add spinach, nuts, cheese and cream to risotto; cook, stirring until spinach wilts. Add pumpkin; stir gently into risotto.
\end{enumerate}
